    levelInputs.forEach(input => input.addEventListener('change', filterByLevel));

    /**
     * 判断输入项所在的能力等级行当前是否已经解锁。
     *
     * 条件面板本身可以暂时隐藏，但其中的默认值仍需保留；这里只判断等级过滤产生的
     * d-none，避免用户再次打开疑问句或感叹句面板时出现没有默认选项的空状态。
     *
     * @param {HTMLInputElement} input 需要检查的单选或复选输入项。
     * @returns {boolean} 当前能力等级是否允许使用该输入项。
     */
    function isInputUnlocked(input) {
        const row = input.closest('.option-row[data-level]');
        return Boolean(row && !row.classList.contains('d-none'));
    }

    /**
     * 为所有已经解锁的单选组补齐默认值。
     *
     * 降低等级会清除超出能力范围的选项；重新提高等级时，如果某个单选组没有值，
     * 就选中该组第一个已解锁项，确保特殊疑问词和感叹结构始终拥有合法默认值。
     *
     * @returns {void}
     */
    function ensureUnlockedRadioDefaults() {
        const radios = Array.from(document.querySelectorAll('input[type="radio"]:not([name="user_level"])'));
        const radioNames = new Set(radios.map(radio => radio.name));

        radioNames.forEach(name => {
            const group = radios.filter(radio => radio.name === name);
            const hasUnlockedSelection = group.some(radio => radio.checked && isInputUnlocked(radio));
            if (hasUnlockedSelection) return;

            const fallback = group.find(isInputUnlocked);
            if (fallback) fallback.checked = true;
        });
    }

    /**
     * 同步一个区域的视觉禁用、键盘可操作性与无障碍状态。
     *
     * @param {HTMLElement} section 需要启用或禁用的区域容器。
     * @param {boolean} enabled true 表示允许交互，false 表示整个区域不适用。
     * @returns {void}
     */
    function setControlSectionEnabled(section, enabled) {
        section.classList.toggle('disabled-layer', !enabled);
        section.setAttribute('aria-disabled', String(!enabled));
        section.querySelectorAll('input').forEach(input => {
            input.disabled = !enabled;
        });
    }

    /**
     * 单独切换一个选项的可用状态。
     *
     * 整块区域是否可用由 setControlSectionEnabled 处理；这个方法只负责区域内部
     * 确定不适用于当前句型的单个选项，例如祈使句中的 There be。
     *
     * @param {string} inputId 需要切换状态的 input 元素 ID。
     * @param {boolean} enabled true 表示允许选择，false 表示锁定该选项。
     * @returns {void}
     */
    function setControlOptionEnabled(inputId, enabled) {
        const input = document.getElementById(inputId);
        if (!input) return;

        const row = input.closest('.option-row');
        input.disabled = !enabled;
        if (!row) return;

        row.classList.toggle('disabled-layer', !enabled);
        row.setAttribute('aria-disabled', String(!enabled));
    }

    /**
     * 读取页面内部判断规则使用的原始值。
     *
     * value 是页面自己的短标识，只用于 JavaScript 判断；data-prompt-value 是发给
     * AI 的完整说明。两者分开后，AI 参数可以写得很清楚，又不会影响现有联动规则。
     *
     * @param {string} name input 的 name 分组名。
     * @param {boolean} includeDisabled 是否允许读取暂时被规则禁用的选项。
     * @returns {string|null} 页面内部值；没有合法选项时返回 null。
     */
    function getSelectedControlValue(name, includeDisabled = false) {
        const checked = Array.from(document.querySelectorAll(`input[name="${name}"]:checked`))
            .find(input => isInputUnlocked(input) && (includeDisabled || !input.disabled));
        return checked ? checked.value : null;
    }

    /**
     * 判断当前是否选择了会覆盖常规时态的虚拟语气结构。
     *
     * @returns {boolean} 任一虚拟语气结构被选中时返回 true。
     */
    function hasSelectedSubjunctive() {
        return ['as_cond', 'as_req', 'as_wish'].some(id => {
            const input = document.getElementById(id);
            return Boolean(input && input.checked && isInputUnlocked(input));
        });
    }

    /**
     * 读取一个单选组当前选项的稳定 Prompt 值。
     *
     * 页面展示文字只服务用户；发送给 AI 的值优先读取 data-prompt-value，避免以后
     * 修改中文标签时意外改变接口数据。最终 JSON 默认排除被当前规则禁用的选项。
     *
     * @param {string} name input 的 name 分组名。
     * @param {boolean} includeDisabled 是否允许读取当前被规则禁用的选项。
     * @returns {string|null} 稳定 Prompt 值；没有合法选项时返回 null。
     */
    function getSelectedPromptValue(name, includeDisabled = false) {
        const checked = Array.from(document.querySelectorAll(`input[name="${name}"]:checked`))
            .find(input => isInputUnlocked(input) && (includeDisabled || !input.disabled));
        return checked ? (checked.dataset.promptValue || checked.value) : null;
    }

    /**
     * 读取一个复选组中所有有效选项的稳定 Prompt 值。
     *
     * @param {string} name input 的 name 分组名。
     * @returns {string[]} 保持页面顺序的 Prompt 值列表。
     */
    function getSelectedPromptValues(name) {
        return Array.from(document.querySelectorAll(`input[name="${name}"]:checked:not(:disabled)`))
            .filter(isInputUnlocked)
            .map(input => input.dataset.promptValue || input.value);
    }

    /**
     * 根据句式和基本骨架计算当前真正生效的语法区域。
     *
     * 这份结果同时驱动 UI 与 AI JSON，是互斥规则的唯一来源：感叹句由 What/How
     * 结构接管基础骨架；祈使句使用隐含主语 you；There be 使用虚设主语 there；
     * SV 与 There be 都不读取普通宾语/补语选项。
     *
     * @returns {object} 当前选项经过语法互斥后的规则状态。
     */
    function resolveGrammarRules() {
        const sentenceType = getSelectedControlValue('sentType', true) || 'declarative';
        const pattern = getSelectedControlValue('pattern', true) || 'sv';
        const isExclamatory = sentenceType === 'exclamatory';
        const isImperative = sentenceType === 'imperative';
        const usesThereBe = !isExclamatory && pattern === 'therebe';
        const hasObjectOrComplement = !['sv', 'therebe'].includes(pattern);

        return {
            sentenceType,
            pattern,
            isExclamatory,
            isImperative,
            usesThereBe,
            patternEnabled: !isExclamatory,
            subjectEnabled: !isExclamatory && !isImperative && !usesThereBe,
            objectEnabled: !isExclamatory && hasObjectOrComplement,
        };
    }

    /**
     * 把当前有效状态整理成可直接发送给 AI 的结构化数据。
     *
     * null 明确表示该字段被当前语法规则接管，而不是用户漏选；constraints 进一步
     * 告诉服务端为什么忽略字段，AI 无需从中文界面文案中猜测优先级。
     *
    * @returns {object} 经过互斥处理、字段稳定且可 JSON.stringify 的请求载荷。
     */
    function buildAiPayload() {
        // API 层可能绕过页面点击直接读取数据，因此生成前再次执行确定冲突清理。
        normalizeCertainConflicts();

        const rules = resolveGrammarRules();
        const isSubjunctive = hasSelectedSubjunctive();
        const levelInput = document.querySelector('input[name="user_level"]:checked');
        const level = Number(levelInput ? levelInput.value : 0);
        const constraints = [];

        let subject;
        if (rules.isExclamatory) {
            subject = { mode: 'controlled_by_exclamatory_structure', value: null };
            constraints.push('exclamatory_structure_controls_base_syntax');
        } else if (rules.isImperative) {
            subject = { mode: 'implicit_imperative_subject', value: 'you' };
            constraints.push(
                'omit_explicit_subject',
                'use_base_verb',
                'do_not_use_modal_auxiliary_in_imperative_main_clause'
            );
        } else if (rules.usesThereBe) {
            subject = { mode: 'existential_expletive_subject', value: 'there' };
            constraints.push('use_existential_there_be', 'generate_existential_content');
        } else {
            subject = {
                mode: 'user_selected_subject_type',
                value: getSelectedPromptValue('subj_type'),
            };
        }

        if (rules.pattern === 'sv' && !rules.isExclamatory) {
            constraints.push('intransitive_pattern_has_no_object');
        }
        if (isSubjunctive && !rules.isImperative) {
            constraints.push('subjunctive_controls_tense');
        }

        const interrogativeType = rules.sentenceType === 'interrogative'
            ? getSelectedControlValue('intType')
            : null;
        const questionWord = interrogativeType === 'special'
            ? getSelectedPromptValue('whWord')
            : null;
        const exclamativeType = rules.isExclamatory
            ? getSelectedPromptValue('exclType')
            : null;
        const tenseSource = rules.isImperative
            ? 'imperative_base_form'
            : (isSubjunctive
                ? 'controlled_by_selected_subjunctive_structure'
                : 'user_selected_tense');

        return {
            schema_version: 1,
            request: {
                task: 'generate_one_grammatically_correct_english_sentence',
                output: 'return_only_the_generated_english_sentence',
            },
            learner_level: {
                score: level,
                scale: '0_beginner_to_1000_advanced',
                instruction: 'match_vocabulary_and_sentence_complexity_to_this_score',
            },
            sentence_type: {
                kind: getSelectedPromptValue('sentType', true),
                subtype: interrogativeType
                    ? getSelectedPromptValue('intType')
                    : exclamativeType,
                question_word: questionWord,
            },
            syntax: {
                pattern: rules.patternEnabled ? getSelectedPromptValue('pattern') : null,
                subject,
                object_or_complement: rules.objectEnabled
                    ? getSelectedPromptValue('obj_type')
                    : null,
            },
            grammar: {
                voice: getSelectedPromptValue('voice'),
                polarity: rules.isExclamatory
                    ? 'affirmative_polarity'
                    : getSelectedPromptValue('polarity'),
                tense: rules.isImperative || isSubjunctive
                    ? null
                    : getSelectedPromptValue('tense'),
                tense_source: tenseSource,
            },
            modifiers: {
                modals: getSelectedPromptValues('ext_modals'),
                attributes: getSelectedPromptValues('ext_attr'),
                adverbials: getSelectedPromptValues('ext_adv'),
                noun_clauses: getSelectedPromptValues('ext_noun_clause'),
                coordinations: getSelectedPromptValues('ext_coord'),
                advanced_structures: getSelectedPromptValues('ext_adv_struc'),
            },
            constraints,
        };
    }

    function filterByLevel() {
        const selectedLevel = parseInt(document.querySelector('input[name="user_level"]:checked').value, 10);
        
        // 更新顶部 Badge 标识
        document.getElementById('current-level-badge').innerText = levelNamesMap[selectedLevel] || `${selectedLevel}分`;

        // 包含式过滤所有包含 data-level 属性的行
        const rows = document.querySelectorAll('.option-row[data-level]');
        rows.forEach(row => {
            const itemLevel = parseInt(row.getAttribute('data-level'), 10);
            const rowInput = row.querySelector('input');

            if (itemLevel <= selectedLevel) {
                row.classList.remove('d-none');
            } else {
                row.classList.add('d-none');
                if (rowInput && rowInput.checked) {
                    rowInput.checked = false;
                }
            }
        });

        // 所有行完成显示状态更新后，再统一补齐单选组，避免循环中的前后顺序影响结果。
        ensureUnlockedRadioDefaults();
        updateMobileLogic();
    }

    // 核心 UI 逻辑联动
    const allInputs = document.querySelectorAll('input:not([name="user_level"])');
    allInputs.forEach(input => input.addEventListener('change', updateMobileLogic));

    // 核心情态动词共享同一个限定动词位置，同一句主干只能选择其中一个。
    // 半情态结构与 Modal + have done 可以和核心情态组合，因此不放进互斥集合。
    const coreModalIds = ['m_can', 'm_will', 'm_must', 'm_shall', 'm_may', 'm_ought', 'm_need'];
    const coreModalInputs = coreModalIds.map(id => document.getElementById(id));
    coreModalInputs.forEach(current => {
        current.addEventListener('change', () => {
            // 取消当前项表示“不使用核心情态”，此时无需影响其他结构。
            if (!current.checked) return;
            coreModalInputs.forEach(other => {
                if (other !== current) other.checked = false;
            });
        });
    });

    // 一键清空修饰
    document.getElementById('reset-ext-btn').addEventListener('click', function() {
        const extCheckboxes = document.querySelectorAll('input[name^="ext_"]');
        extCheckboxes.forEach(cb => cb.checked = false);
        updateMobileLogic();
    });

    /**
     * 清理已经确定无法同时成立的选择。
     *
     * 这里只处理没有语法争议的冲突：普通祈使句不能以 There be 作为主干，也不能
     * 在主句谓语前再放情态助动词；形式宾语和使役感官补语则都需要“宾语+补语”骨架。
     * 少见但仍可能正确的组合不会在这里被清除。
     *
     * @returns {void}
     */
    function normalizeCertainConflicts() {
        const sentenceType = getSelectedControlValue('sentType', true) || 'declarative';

        if (sentenceType === 'imperative') {
            const thereBePattern = document.getElementById('pat_therebe');
            if (thereBePattern.checked) {
                document.getElementById('pat_sv').checked = true;
            }

            document.querySelectorAll('input[name="ext_modals"]').forEach(input => {
                input.checked = false;
            });
        }

        const pattern = getSelectedControlValue('pattern', true) || 'sv';
        if (pattern !== 'svoc') {
            const objectComplementOptions = ['obj_dummy', 'obj_causative']
                .map(id => document.getElementById(id));
            if (objectComplementOptions.some(input => input.checked)) {
                document.getElementById('obj_noun').checked = true;
            }
        }
    }

    function updateMobileLogic() {
        // 先移除确定冲突的旧选择，再用清理后的状态计算本轮页面联动。
        normalizeCertainConflicts();

        // 先解析一次有效规则，后续每个区域都使用同一份判断结果。
        const rules = resolveGrammarRules();
        const pattern = rules.pattern;
        const sentType = rules.sentenceType;

        // 感叹句忽略基础骨架；祈使句与 There be 则分别接管主语构成。
        setControlSectionEnabled(document.getElementById('pattern-section'), rules.patternEnabled);
        setControlSectionEnabled(document.getElementById('subject-section'), rules.subjectEnabled);
        setControlSectionEnabled(document.getElementById('object-section'), rules.objectEnabled);

        // 祈使句使用隐含的 you + 动词原形，不能再使用普通 There be 主干或情态助动词。
        setControlOptionEnabled('pat_therebe', rules.patternEnabled && !rules.isImperative);
        document.querySelectorAll('input[name="ext_modals"]').forEach(input => {
            setControlOptionEnabled(input.id, !rules.isImperative);
        });

        // 形式宾语和使役感官结构都包含“宾语+补语”，只对 S+V+O+C 骨架开放。
        const acceptsObjectComplement = rules.objectEnabled && pattern === 'svoc';
        setControlOptionEnabled('obj_dummy', acceptsObjectComplement);
        setControlOptionEnabled('obj_causative', acceptsObjectComplement);

        // 逻辑 1：S+V, S+V+P, There Be 不适用被动语态
        const passiveRadio = document.getElementById('voice_pas');
        const activeRadio = document.getElementById('voice_act');
        const passiveRow = document.getElementById('voice-passive-row');
        const voiceWarning = document.getElementById('voice-warning');

        // 感叹句不使用基础骨架，所以旧骨架不能继续限制其语态。
        if (rules.patternEnabled && ['sv', 'svp', 'therebe'].includes(pattern)) {
            passiveRow.classList.add('disabled-layer');
            passiveRadio.disabled = true;
            activeRadio.checked = true;
            voiceWarning.classList.remove('d-none');
        } else {
            passiveRow.classList.remove('disabled-layer');
            passiveRadio.disabled = false;
            voiceWarning.classList.add('d-none');
        }

        // 逻辑 2：疑问句与感叹句面板联动
        const interBox = document.getElementById('interrogative-options');
        const exclBox = document.getElementById('exclamatory-options');
        const whBox = document.getElementById('wh-words-box');
        const polarityGroup = document.getElementById('polarity-group');
        const polarityTitle = document.getElementById('polarity-title');
        const affirmativeRadio = document.getElementById('pol_aff');

        interBox.classList.add('d-none');
        exclBox.classList.add('d-none');
        whBox.classList.add('d-none');

        if (sentType === 'interrogative') {
            interBox.classList.remove('d-none');
            const intType = document.querySelector('input[name="intType"]:checked');
            if (intType && intType.value === 'special') {
                whBox.classList.remove('d-none');
            }
        } else if (sentType === 'exclamatory') {
            exclBox.classList.remove('d-none');
        }

        if (sentType === 'exclamatory') {
            // 感叹结构不读取普通极性；先复位为肯定，避免保留并输出旧的“否定句”状态。
            affirmativeRadio.checked = true;
            setControlSectionEnabled(polarityGroup, false);
            polarityTitle.classList.add('text-muted');
        } else {
            setControlSectionEnabled(polarityGroup, true);
            polarityTitle.classList.remove('text-muted');
        }

        // 逻辑 3：祈使句无常规时态，虚拟语气结构会覆盖常规时态选择。
        const tenseBody = document.getElementById('tense-body');
        const tenseWarning = document.getElementById('tense-warning');
        const subjWarning = document.getElementById('subjunctive-warning');
        const isSubjunctive = hasSelectedSubjunctive();
        const isImperative = sentType === 'imperative';

        setControlSectionEnabled(tenseBody, !isImperative && !isSubjunctive);
        tenseWarning.classList.toggle('d-none', !isImperative);
        // 祈使句提示优先，避免同一个标题区域同时出现两个互相竞争的原因。
        subjWarning.classList.toggle('d-none', isImperative || !isSubjunctive);
    }

    /**
     * 把文本写入系统剪贴板，并兼容直接通过 file:// 打开的页面。
     *
     * 安全上下文优先使用现代 Clipboard API；本地文件或旧浏览器没有权限时，临时
     * 创建屏幕外 textarea 并执行 copy 命令，完成后立即移除，不污染页面内容。
     *
     * @param {string} text 需要完整复制的预览文字或格式化 JSON。
     * @returns {Promise<void>} 复制成功时完成，两个方案都失败时抛出异常。
     */
    async function copyTextToClipboard(text) {
        if (!text) throw new Error('没有可复制的内容');

        if (navigator.clipboard && window.isSecureContext) {
            try {
                await navigator.clipboard.writeText(text);
                return;
            } catch (error) {
                // 用户拒绝权限时继续尝试本地文件也可用的兼容方案。
            }
        }

        const textarea = document.createElement('textarea');
        textarea.value = text;
        textarea.readOnly = true;
        textarea.style.position = 'fixed';
        textarea.style.left = '-9999px';
        textarea.style.opacity = '0';
        // 弹窗打开时放进弹窗内部，避免 Bootstrap 焦点锁把 textarea 的焦点夺走。
        const copyHost = document.querySelector('.modal.show') || document.body;
        copyHost.appendChild(textarea);
        textarea.focus();
        textarea.select();
        textarea.setSelectionRange(0, textarea.value.length);
        const copied = document.execCommand('copy');
        textarea.remove();
        if (!copied) throw new Error('浏览器拒绝复制');
    }

    // WeakMap 以按钮节点为键保存复位计时器，重复点击时不会被旧计时器提前还原。
    const copyFeedbackTimers = new WeakMap();

    /**
     * 用图标、颜色和无障碍文本反馈一次复制结果，然后恢复默认状态。
     *
     * @param {HTMLButtonElement} button 触发复制的按钮。
     * @param {boolean} succeeded 本次复制是否成功。
     * @returns {void}
     */
    function showCopyFeedback(button, succeeded) {
        const defaultLabel = button.dataset.copyLabel || '复制';
        const resultLabel = succeeded ? '已复制' : '复制失败';
        const idleIcon = button.querySelector('.copy-idle-icon');
        const successIcon = button.querySelector('.copy-success-icon');
        const oldTimer = copyFeedbackTimers.get(button);

        if (oldTimer) window.clearTimeout(oldTimer);
        idleIcon.classList.toggle('d-none', succeeded);
        successIcon.classList.toggle('d-none', !succeeded);
        button.classList.remove('btn-outline-secondary', 'btn-outline-success', 'btn-outline-danger');
        button.classList.add(succeeded ? 'btn-outline-success' : 'btn-outline-danger');
        button.setAttribute('aria-label', resultLabel);
        button.title = resultLabel;
        document.getElementById('recipe-copy-status').textContent = resultLabel;

        const resetTimer = window.setTimeout(() => {
            idleIcon.classList.remove('d-none');
            successIcon.classList.add('d-none');
            button.classList.remove('btn-outline-success', 'btn-outline-danger');
            button.classList.add('btn-outline-secondary');
            button.setAttribute('aria-label', defaultLabel);
            button.title = defaultLabel;
            copyFeedbackTimers.delete(button);
        }, 1600);
        copyFeedbackTimers.set(button, resetTimer);
    }

    // 两个标签页共用同一套复制流程，只通过 data-copy-target 指定内容节点。
    document.querySelectorAll('[data-copy-target]').forEach(button => {
        button.addEventListener('click', async () => {
            const target = document.querySelector(button.dataset.copyTarget);
            // innerText 会保留预览中的可见换行，JSON 的缩进和换行也会原样保留。
            const text = target ? target.innerText : '';
            button.disabled = true;
            try {
                await copyTextToClipboard(text);
                showCopyFeedback(button, true);
            } catch (error) {
                showCopyFeedback(button, false);
            } finally {
                button.disabled = false;
            }
        });
    });

    // 生成配方按钮逻辑
    document.getElementById('generate-btn').addEventListener('click', function() {
        // AI 数据与人类预览在同一次点击中生成，确保两者描述的是同一份有效状态。
        const aiPayload = buildAiPayload();
        const rules = resolveGrammarRules();

        const getRadioText = (name) => {
            const checked = document.querySelector(`input[name="${name}"]:checked:not(:disabled)`);
            if (!checked) return '无';
            const row = checked.closest('.option-row');
            if (!row || row.classList.contains('d-none')) return '无';
            const label = row ? row.querySelector('.form-check-label') : null;
            return label ? label.innerText.trim() : '无';
        };

        const getCheckboxesText = (name) => {
            const checked = Array.from(document.querySelectorAll(`input[name="${name}"]:checked:not(:disabled)`))
                .filter(input => isInputUnlocked(input));
            if (checked.length === 0) return null;
            return checked.map(c => {
                const row = c.closest('.option-row');
                const label = row ? row.querySelector('.form-check-label') : null;
                return label ? label.innerText.trim() : '';
            }).filter(Boolean).join(', ');
        };

        let summary = [];
        const levelName = document.getElementById('current-level-badge').innerText;
        const patternText = rules.isExclamatory
            ? '由 What / How 感叹结构决定'
            : getRadioText('pattern');
        const subjectText = rules.isExclamatory
            ? '由感叹结构决定'
            : (rules.isImperative
                ? '隐含主语 You'
                : (rules.usesThereBe
                    ? '虚设主语 There'
                    : getRadioText('subj_type')));
        const objectText = rules.isExclamatory
            ? '由感叹结构决定'
            : (rules.pattern === 'sv'
                ? '不适用（该骨架无宾语）'
                : (rules.usesThereBe
                    ? '存在内容由 AI 决定'
                    : getRadioText('obj_type')));

        summary.push(`【能力阶段】: ${levelName}`);
        summary.push(`【基本骨架】: ${patternText}`);
        summary.push(`【主宾构成】: 主语(${subjectText}) | 宾/补语(${objectText})`);
        
        const currentSentType = rules.sentenceType;
        const polarityText = rules.isExclamatory
            ? '固定为肯定形式（感叹结构）'
            : getRadioText('polarity');
        summary.push(`【语态极性】: ${getRadioText('voice')} | ${polarityText}`);

        let sentTypeStr = getRadioText('sentType');
        if (currentSentType === 'interrogative') {
            sentTypeStr += ` -> ${getRadioText('intType')}`;
            const intTypeChecked = document.querySelector('input[name="intType"]:checked');
            if (intTypeChecked && intTypeChecked.value === 'special') {
                sentTypeStr += ` (${getRadioText('whWord')})`;
            }
        } else if (currentSentType === 'exclamatory') {
            sentTypeStr += ` -> ${getRadioText('exclType')}`;
        }
        summary.push(`【沟通句式】: ${sentTypeStr}`);
        
        if (currentSentType === 'imperative') {
            summary.push(`【时态选择】: 无 (祈使句动词原形)`);
        } else if (hasSelectedSubjunctive()) {
            summary.push(`【时态选择】: 由所选虚拟语气结构决定`);
        } else {
            summary.push(`【时态选择】: ${getRadioText('tense')}`);
        }

        let extDetails = [];
        if (getCheckboxesText('ext_modals')) extDetails.push(`情绪/情态: ${getCheckboxesText('ext_modals')}`);
        if (getCheckboxesText('ext_attr')) extDetails.push(`名词定语: ${getCheckboxesText('ext_attr')}`);
        if (getCheckboxesText('ext_adv')) extDetails.push(`背景状语: ${getCheckboxesText('ext_adv')}`);
        if (getCheckboxesText('ext_noun_clause')) extDetails.push(`名词从句: ${getCheckboxesText('ext_noun_clause')}`);
        if (getCheckboxesText('ext_coord')) extDetails.push(`并列复合: ${getCheckboxesText('ext_coord')}`);
        if (getCheckboxesText('ext_adv_struc')) extDetails.push(`高阶句法: ${getCheckboxesText('ext_adv_struc')}`);

        if (extDetails.length > 0) {
            summary.push(`\n【高阶修饰层】:\n - ` + extDetails.join('\n - '));
        } else {
            summary.push(`\n【高阶修饰层】: 未叠加额外修饰`);
        }

        const recipeOutput = document.getElementById('recipe-output');
        recipeOutput.innerText = summary.join('\n');
        // 紧凑 JSON 留给后续请求逻辑读取，格式化 JSON 则写入用户可检查的代码页。
        const serializedPayload = JSON.stringify(aiPayload);
        recipeOutput.dataset.aiPayload = serializedPayload;
        document.getElementById('recipe-json-output').textContent = JSON.stringify(aiPayload, null, 2);
        
        const modalEl = document.getElementById('recipe-modal');
        // 每次生成默认回到人类预览，用户需要时再切换到 AI JSON。
        const previewTab = document.getElementById('recipe-preview-tab');
        bootstrap.Tab.getOrCreateInstance(previewTab).show();
        const modalInstance = bootstrap.Modal.getOrCreateInstance(modalEl);
        modalInstance.show();
    });

    // 暴露只读构建入口，后续 API 层可直接调用 SentenceForge.buildAiPayload()。
    window.SentenceForge = Object.freeze({ buildAiPayload });

    // 初始化运行
    initMockMasteryData();
    filterByLevel();
</script>
